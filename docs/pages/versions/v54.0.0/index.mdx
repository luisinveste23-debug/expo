import React, { useState, useEffect } from "react";
import { View, Text, TextInput, Button, FlatList, Alert, ScrollView } from "react-native";
import AsyncStorage from "@react-native-async-storage/async-storage";
import * as FileSystem from "expo-file-system";
import * as Sharing from "expo-sharing";
import * as Print from "expo-print";

export default function App() {
  const [talhao, setTalhao] = useState("");
  const [dose, setDose] = useState("");
  const [unidade, setUnidade] = useState("");
  const [vazao, setVazao] = useState("");
  const [data, setData] = useState("");
  const [observacao, setObservacao] = useState("");
  const [aplicacoes, setAplicacoes] = useState([]);

  useEffect(() => {
    carregarDados();
  }, []);

  const carregarDados = async () => {
    const dataSalva = await AsyncStorage.getItem("aplicacoes");
    if (dataSalva) setAplicacoes(JSON.parse(dataSalva));
  };

  const salvarAplicacao = async () => {
    if (!talhao || !dose || !data) {
      Alert.alert("Campos obrigatórios", "Preencha talhão, dose e data!");
      return;
    }
    const nova = { id: Date.now().toString(), talhao, dose, unidade, vazao, data, observacao };
    const atualizadas = [...aplicacoes, nova];
    setAplicacoes(atualizadas);
    await AsyncStorage.setItem("aplicacoes", JSON.stringify(atualizadas));
    limparCampos();
  };

  const limparCampos = () => {
    setTalhao(""); setDose(""); setUnidade(""); setVazao(""); setData(""); setObservacao("");
  };

  const exportarCSV = async () => {
    if (aplicacoes.length === 0) return Alert.alert("Nada para exportar!");
    const csv = "Talhão,Data,Dose,Unidade,Vazão,Observação\n" +
      aplicacoes.map(a => `${a.talhao},${a.data},${a.dose},${a.unidade},${a.vazao},${a.observacao}`).join("\n");
    const caminho = FileSystem.documentDirectory + "aplicacoes.csv";
    await FileSystem.writeAsStringAsync(caminho, csv, { encoding: FileSystem.EncodingType.UTF8 });
    await Sharing.shareAsync(caminho);
  };

  const exportarPDF = async () => {
    if (aplicacoes.length === 0) return Alert.alert("Nada para exportar!");
    let html = `
      <h1>Relatório de Aplicações</h1>
      <table border="1" style="width:100%; border-collapse:collapse;">
      <tr><th>Talhão</th><th>Data</th><th>Dose</th><th>Unidade</th><th>Vazão</th><th>Obs</th></tr>
      ${aplicacoes.map(a => `<tr><td>${a.talhao}</td><td>${a.data}</td><td>${a.dose}</td><td>${a.unidade}</td><td>${a.vazao}</td><td>${a.observacao}</td></tr>`).join("")}
      </table>
    `;
    await Print.printAsync({ html });
  };

  const limparTudo = async () => {
    await AsyncStorage.removeItem("aplicacoes");
    setAplicacoes([]);
  };

  return (
    <ScrollView style={{ padding: 20, marginTop: 40 }}>
      <Text style={{ fontSize: 24, fontWeight: "bold", marginBottom: 10 }}>🌾 Agro Control Aplicações</Text>

      <TextInput placeholder="Talhão" value={talhao} onChangeText={setTalhao} style={{ borderWidth: 1, marginBottom: 5, padding: 8 }} />
      <TextInput placeholder="Data (ex: 03/11/2025)" value={data} onChangeText={setData} style={{ borderWidth: 1, marginBottom: 5, padding: 8 }} />
      <TextInput placeholder="Dose" value={dose} onChangeText={setDose} style={{ borderWidth: 1, marginBottom: 5, padding: 8 }} />
      <TextInput placeholder="Unidade" value={unidade} onChangeText={setUnidade} style={{ borderWidth: 1, marginBottom: 5, padding: 8 }} />
      <TextInput placeholder="Vazão" value={vazao} onChangeText={setVazao} style={{ borderWidth: 1, marginBottom: 5, padding: 8 }} />
      <TextInput placeholder="Observação" value={observacao} onChangeText={setObservacao} style={{ borderWidth: 1, marginBottom: 10, padding: 8 }} />

      <Button title="Salvar Aplicação" onPress={salvarAplicacao} />
      <View style={{ marginVertical: 10 }} />
      <Button title="Exportar CSV" onPress={exportarCSV} />
      <View style={{ marginVertical: 10 }} />
      <Button title="Exportar PDF" onPress={exportarPDF} />
      <View style={{ marginVertical: 10 }} />
      <Button title="Limpar Tudo" color="red" onPress={limparTudo} />

      <Text style={{ fontSize: 18, marginTop: 20, fontWeight: "bold" }}>📋 Aplicações Salvas:</Text>
      <FlatList
        data={aplicacoes}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <View style={{ borderBottomWidth: 1, padding: 5 }}>
            <Text>🌱 Talhão: {item.talhao}</Text>
            <Text>📅 Data: {item.data}</Text>
            <Text>💧 Dose: {item.dose} {item.unidade}</Text>
            <Text>🚜 Vazão: {item.vazao}</Text>
            <Text>🗒️ Obs: {item.observacao}</Text>
          </View>
        )}
      />
    </ScrollView>
  );
}
