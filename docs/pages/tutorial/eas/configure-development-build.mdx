// CampoFácil - App básico para controle de aplicações
// Funcionalidades: múltiplos produtos, salvar histórico, gerar PDF com marca d'água "Feito por Felipe Ferreira"
// Testado para rodar via Expo Go (iOS e Android)

import React, { useState, useEffect } from "react";
import {
  View,
  Text,
  TextInput,
  Button,
  FlatList,
  TouchableOpacity,
  Alert,
  ScrollView,
  StyleSheet,
  Platform,
} from "react-native";
import AsyncStorage from "@react-native-async-storage/async-storage";
import * as Print from "expo-print";
import * as Sharing from "expo-sharing";

const STORAGE_KEY = "@campofacil_aplicacoes_v1";

function ProdutoRow({ item, onChange, onRemove }) {
  return (
    <View style={styles.prodRow}>
      <TextInput
        placeholder="Produto"
        value={item.produto}
        onChangeText={(t) => onChange("produto", t)}
        style={[styles.input, styles.prodInput]}
      />
      <TextInput
        placeholder="Dose L/ha"
        value={item.doseLha}
        onChangeText={(t) => onChange("doseLha", t)}
        style={[styles.input, styles.smallInput]}
        keyboardType="numeric"
      />
      <TextInput
        placeholder="Dose L/bomba"
        value={item.doseLbomba}
        onChangeText={(t) => onChange("doseLbomba", t)}
        style={[styles.input, styles.smallInput]}
        keyboardType="numeric"
      />
      <TextInput
        placeholder="Obs"
        value={item.obs}
        onChangeText={(t) => onChange("obs", t)}
        style={[styles.input, styles.prodInput]}
      />
      <TouchableOpacity onPress={onRemove} style={styles.removeBtn}>
        <Text style={{ color: "white" }}>Rem</Text>
      </TouchableOpacity>
    </View>
  );
}

export default function App() {
  // ficha fields
  const [pulverizador, setPulverizador] = useState("");
  const [fazenda, setFazenda] = useState("");
  const [talhao, setTalhao] = useState("");
  const [area, setArea] = useState("");
  const [data, setData] = useState(""); // usuário digita data
  const [vazao, setVazao] = useState("");
  const [hectaresBomba, setHectaresBomba] = useState("");
  const [tipoOp, setTipoOp] = useState("");
  const [operador, setOperador] = useState("");

  // produtos dinâmicos
  const [produtos, setProdutos] = useState([
    { id: String(Date.now()), produto: "", doseLha: "", doseLbomba: "", obs: "" },
  ]);

  // histórico
  const [historico, setHistorico] = useState([]);

  useEffect(() => {
    carregarHistorico();
  }, []);

  async function carregarHistorico() {
    try {
      const raw = await AsyncStorage.getItem(STORAGE_KEY);
      if (raw) {
        setHistorico(JSON.parse(raw));
      }
    } catch (e) {
      console.error(e);
    }
  }

  function addProduto() {
    setProdutos((p) => [...p, { id: String(Date.now()), produto: "", doseLha: "", doseLbomba: "", obs: "" }]);
  }

  function updateProduto(index, key, value) {
    setProdutos((p) => {
      const copy = [...p];
      copy[index] = { ...copy[index], [key]: value };
      return copy;
    });
  }

  function removeProduto(index) {
    setProdutos((p) => p.filter((_, i) => i !== index));
  }

  async function salvarAplicacao() {
    if (!fazenda || !talhao || !data) {
      Alert.alert("Preencha os campos", "Fazenda, Talhão e Data são obrigatórios.");
      return;
    }

    const item = {
      id: String(Date.now()),
      pulverizador,
      fazenda,
      talhao,
      area,
      data,
      vazao,
      hectaresBomba,
      tipoOp,
      operador,
      produtos: produtos.map((p) => ({ produto: p.produto, doseLha: p.doseLha, doseLbomba: p.doseLbomba, obs: p.obs })),
      createdAt: new Date().toISOString(),
    };

    const novoHist = [item, ...historico];
    setHistorico(novoHist);
    await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(novoHist));
    Alert.alert("Salvo", "Aplicação salva no histórico.");
    limparFormulario();
  }

  function limparFormulario() {
    setPulverizador("");
    setFazenda("");
    setTalhao("");
    setArea("");
    setData("");
    setVazao("");
    setHectaresBomba("");
    setTipoOp("");
    setOperador("");
    setProdutos([{ id: String(Date.now()), produto: "", doseLha: "", doseLbomba: "", obs: "" }]);
  }

  async function gerarPdf(item) {
    try {
      // monta HTML do relatório
      const produtosHtml = (item.produtos || [])
        .map(
          (p, i) =>
            `<tr><td style="padding:6px;border:1px solid #ddd">${i + 1}</td><td style="padding:6px;border:1px solid #ddd">${p.produto || ""}</td><td style="padding:6px;border:1px solid #ddd">${p.doseLha || ""}</td><td style="padding:6px;border:1px solid #ddd">${p.doseLbomba || ""}</td><td style="padding:6px;border:1px solid #ddd">${p.obs || ""}</td></tr>`
        )
        .join("");

      const html = `
        <html>
          <head>
            <meta name="viewport" content="width=device-width, initial-scale=1.0"/>
          </head>
          <body style="font-family: Arial, Helvetica, sans-serif; padding:16px;">
            <h2 style="text-align:center;">CampoFácil - Ficha de Aplicação</h2>
            <p><strong>Fazenda:</strong> ${item.fazenda || ""} &nbsp;&nbsp; <strong>Talhão:</strong> ${item.talhao || ""} &nbsp;&nbsp; <strong>Área (ha):</strong> ${item.area || ""}</p>
            <p><strong>Data:</strong> ${item.data || ""} &nbsp;&nbsp; <strong>Vazão L/ha:</strong> ${item.vazao || ""} &nbsp;&nbsp; <strong>Hectares/Bomba:</strong> ${item.hectaresBomba || ""}</p>
            <p><strong>Pulverizador:</strong> ${item.pulverizador || ""} &nbsp;&nbsp; <strong>Tipo:</strong> ${item.tipoOp || ""} &nbsp;&nbsp; <strong>Operador:</strong> ${item.operador || ""}</p>

            <h3>Produtos</h3>
            <table style="width:100%; border-collapse:collapse; font-size:12px;">
              <thead>
                <tr>
                  <th style="padding:6px;border:1px solid #ddd">#</th>
                  <th style="padding:6px;border:1px solid #ddd">Produto</th>
                  <th style="padding:6px;border:1px solid #ddd">Dose L/ha</th>
                  <th style="padding:6px;border:1px solid #ddd">Dose L/bomba</th>
                  <th style="padding:6px;border:1px solid #ddd">Observação</th>
                </tr>
              </thead>
              <tbody>
                ${produtosHtml}
              </tbody>
            </table>

            <div style="margin-top:20px; font-size:11px; color:#444">
              <em>Gerado em: ${new Date(item.createdAt).toLocaleString()}</em>
            </div>

            <div style="position:fixed; bottom:10px; width:100%; text-align:center; font-size:10px; color:#999;">
              Feito por Felipe Ferreira
            </div>
          </body>
        </html>
      `;

      // gerar PDF
      const { uri } = await Print.printToFileAsync({ html });
      // compartilhar (mostra opção imprimir/salvar no iOS, compartilhar no Android)
      if (Platform.OS === "android" || Platform.OS === "ios") {
        await Sharing.shareAsync(uri);
      } else {
        // web fallback
        Alert.alert("PDF gerado", `Arquivo: ${uri}`);
      }
    } catch (e) {
      console.error(e);
      Alert.alert("Erro", "Não foi possível gerar o PDF.");
    }
  }

  async function apagarHistorico() {
    Alert.alert("Confirmar", "Apagar todo o histórico?", [
      { text: "Cancelar", style: "cancel" },
      {
        text: "Apagar",
        style: "destructive",
        onPress: async () => {
          await AsyncStorage.removeItem(STORAGE_KEY);
          setHistorico([]);
        },
      },
    ]);
  }

  return (
    <ScrollView contentContainerStyle={{ padding: 16, paddingBottom: 60 }}>
      <Text style={styles.title}>🌾 CampoFácil</Text>

      <Text style={styles.sectionTitle}>Ficha de Aplicação</Text>

      <TextInput placeholder="Pulverizador" value={pulverizador} onChangeText={setPulverizador} style={styles.input} />
      <TextInput placeholder="Fazenda" value={fazenda} onChangeText={setFazenda} style={styles.input} />
      <TextInput placeholder="Talhão" value={talhao} onChangeText={setTalhao} style={styles.input} />
      <TextInput placeholder="Área (ha)" value={area} onChangeText={setArea} style={styles.input} keyboardType="numeric" />
      <TextInput placeholder="Data (ex: 03/11/2025)" value={data} onChangeText={setData} style={styles.input} />
      <TextInput placeholder="Vazão (L/ha)" value={vazao} onChangeText={setVazao} style={styles.input} keyboardType="numeric" />
      <TextInput placeholder="Hectares/Bomba" value={hectaresBomba} onChangeText={setHectaresBomba} style={styles.input} keyboardType="numeric" />
      <TextInput placeholder="Tipo de operação" value={tipoOp} onChangeText={setTipoOp} style={styles.input} />
      <TextInput placeholder="Operador" value={operador} onChangeText={setOperador} style={styles.input} />

      <Text style={[styles.sectionTitle, { marginTop: 12 }]}>Produtos</Text>
      {produtos.map((p, i) => (
        <ProdutoRow
          key={p.id}
          item={p}
          onChange={(key, val) => updateProduto(i, key, val)}
          onRemove={() => removeProduto(i)}
        />
      ))}

      <View style={{ flexDirection: "row", justifyContent: "space-between", marginVertical: 8 }}>
        <Button title="+ Produto" onPress={addProduto} />
        <Button title="Limpar ficha" onPress={limparFormulario} color="#ff8c00" />
      </View>

      <View style={{ marginVertical: 6 }}>
        <Button title="Salvar Aplicação" onPress={salvarAplicacao} />
      </View>

      <Text style={[styles.sectionTitle, { marginTop: 18 }]}>Histórico de Aplicações</Text>
      <View style={{ flexDirection: "row", justifyContent: "space-between", marginBottom: 8 }}>
        <Text style={{ color: "#555" }}>Registros: {historico.length}</Text>
        <Button title="Apagar tudo" onPress={apagarHistorico} color="#d9534f" />
      </View>

      {historico.length === 0 && <Text style={{ color: "#777", marginBottom: 20 }}>Nenhuma aplicação registrada.</Text>}

      <FlatList
        data={historico}
        keyExtractor={(it) => it.id}
        renderItem={({ item }) => (
          <View style={styles.card}>
            <Text style={styles.cardTitle}>{item.fazenda} — {item.talhao}</Text>
            <Text>🗓 {item.data} | Área: {item.area || "-"} ha | Vazão: {item.vazao || "-"}</Text>
            <Text>🚜 Pulverizador: {item.pulverizador || "-" } | Operador: {item.operador || "-"}</Text>
            <Text>Tipo: {item.tipoOp || "-"}</Text>

            <Text style={{ marginTop: 6, fontWeight: "700" }}>Produtos:</Text>
            {item.produtos && item.produtos.length > 0 ? item.produtos.map((p, idx) => (
              <Text key={idx} style={{ fontSize: 13 }}>
                • {p.produto || "-"} — {p.doseLha || "-"} L/ha — {p.doseLbomba || "-"} L/bomba {p.obs ? `(${p.obs})` : ""}
              </Text>
            )) : <Text style={{ color: "#666" }}>— Sem produtos —</Text>}

            <View style={{ flexDirection: "row", justifyContent: "space-between", marginTop: 8 }}>
              <Button title="Gerar PDF" onPress={() => gerarPdf(item)} />
              <Button title="Apagar" color="#d9534f" onPress={async () => {
                const newH = historicalFilterRemove(historico, item.id);
                setHistorico(newH);
                await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(newH));
              }} />
            </View>
          </View>
        )}
      />
    </ScrollView>
  );
}

// helper remove by id
function historicalFilterRemove(list, id) {
  return list.filter(x => x.id !== id);
}

const styles = StyleSheet.create({
  title: { fontSize: 22, fontWeight: "700", marginBottom: 10, textAlign: "center" },
  sectionTitle: { fontSize: 16, fontWeight: "700", marginBottom: 8 },
  input: { backgroundColor: "#fff", padding: 10, marginBottom: 8, borderRadius: 8, borderWidth: 1, borderColor: "#e0e0e0" },
  prodRow: { flexDirection: "row", alignItems: "center", marginBottom: 6, flexWrap: "wrap" },
  prodInput: { flex: 1, minWidth: "42%", marginRight: 6 },
  smallInput: { width: 90, marginRight: 6 },
  removeBtn: { backgroundColor: "#d9534f", padding: 8, borderRadius: 6 },
  card: { backgroundColor: "#fff", padding: 10, marginBottom: 10, borderRadius: 8, borderWidth: 1, borderColor: "#e6e6e6" },
  cardTitle: { fontWeight: "700", marginBottom: 4 },
});
